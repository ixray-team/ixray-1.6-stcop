## Любой класс аномалии наследующий (CAnomalyZone)
### Вертексные партиклы худа гг для аномалий (по мере приближения к центру аномалии на полигонах оружия и детектора проигрываются случайные партиклы)
> [!IMPORTANT]
> **Статус**: Поддерживается <br>
> **Минимальная версия**: 1.5

### Пример настроенной секции
```ini
[pvh_particles_section_default]
use_procedural_vertex_hud_particles = true; Вкл Выкл вертексных худовых партиклов

max_processing_distance = 30; Максимальная дистанция на которой гемплейная фитча получает обновления тиков (оптимизация) учитывать чтобы параметр не был меньше max_procedural_vertex_hud_particles_field_distance иначе частицы будут появляться резко а не линейно по мере бриближения к центру
max_procedural_vertex_hud_particles_field_distance = 20; Максимальная дистанция от центра аномалии на которой начинает воздействовать эффект на худ гг
update_particles_interval_milliseconds = 250; Интервал обновления партиклов (можно считать общей отсрочкой запуска частиц в милисекундах)
identity_group_name = "electric_streams"; Произвольное уникальное имя группы эффектов (не путать с партикл группами) нужно для того чтобы эффекты от нескольких одинаковых аномалий к примеру не складвались на худе (этот же параметр нужен если требуется настроить два и более разных эффектов от разных аномалий находящихся рядом одновременно для худа гг в этом случае нужно указать уникальные имена каждой группе например (огненные, электрические) и тд)

before_play_delay_ms_min = 100; Минимальная задержка в милисекундах перед запуском одной частицы (учитывать еще общий интервал обновлений)
before_play_delay_ms_max = 900; Максимальная задержка в милисекундах перед запуском одной частицы (учитывать еще общий интервал обновлений)

is_allow_particles_for_right_hand = true ;Вкл выкл воздействие частиц на предмет в (правой руке) (выключатель применяется ко всем параметрам правой руки)

right_hand_max_count_particles = 200; Максимальное число партиклов на предмете в правой руке (значение ростёт от 0 (при max_procedural_vertex_hud_particles_field_distance) до указанного по мере приближения гг к центру аномалии)

right_hand_particle_playing_time_ms_min = 450; Минимальное время проигрывания партикла на правой руке
right_hand_particle_playing_time_ms_max = 2350; Максимальное время проигрывания партикла на правой руке

right_hand_after_play_delay_time_ms_min = 200; Минимальное время паузы перед проигрыванием следующего партикла на правой руке
right_hand_after_play_delay_time_ms_max = 800; Максимальное время паузы перед проигрыванием следующего партикла на правой руке

right_hand_particles_paths = ffx0001\electric\streamer_macro_3, ffx0001\electric\streamer_macro_4; Перечисление через запятую путей партиклов для рандомизации на правой руке

is_allow_particles_for_left_hand = true;Вкл выкл воздействие частиц на предмет в (правой руке) (выключатель применяется ко всем параметрам левой руки)

left_hand_max_count_particles = 200 ; Максимальное число партиклов на предмете в левой руке (значение ростёт от 0 (при max_procedural_vertex_hud_particles_field_distance) до указанного по мере приближения гг к центру аномалии)

left_hand_particle_playing_time_ms_min = 450; Минимальное время проигрывания партикла на левой руке
left_hand_particle_playing_time_ms_max = 2350; Максимальное время проигрывания партикла на левой руке

left_hand_after_play_delay_time_ms_min = 200; Минимальное время паузы перед проигрыванием следующего партикла на левой руке
left_hand_after_play_delay_time_ms_max = 800; Максимальное время паузы перед проигрыванием следующего партикла на левой руке

left_hand_particles_paths = ffx0001\electric\streamer_macro_4, ffx0001\electric\streamer_macro_3; Перечисление через запятую путей партиклов для рандомизации на левой руке

; пример конфига аномалии расширяющего zone_witches_galantine_strong
[zone_hud_particles_electra_01]:zone_witches_galantine_strong
procedural_vertex_hud_particles_options_section = pvh_particles_section_default;  Секция описывающая поведение для вертексных худовых партиклов для гг
```
